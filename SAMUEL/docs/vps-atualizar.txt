# SAMUEL — colar o arquivo INTEIRO no SSH (bash).
# Git: /opt/analise
# App: /opt/analise/SAMUEL
# PM2: analise-api (3469) + analise-web (3468)

set -e
cd /opt/analise

STAMP=$(date +%Y%m%d-%H%M%S)
BAK=/root/analise-pre-update-$STAMP
mkdir -p "$BAK"
cp -a SAMUEL/.env "$BAK/.env" 2>/dev/null || true
cp -a SAMUEL/apps/api/.env "$BAK/api.env" 2>/dev/null || true
cp -a SAMUEL/apps/web/.env.local "$BAK/web.env.local" 2>/dev/null || true

git fetch origin
git reset --hard HEAD
git clean -fd
git reset --hard origin/main

echo "HEAD=$(git rev-parse --short HEAD) origin/main=$(git rev-parse --short origin/main)"
test "$(git rev-parse HEAD)" = "$(git rev-parse origin/main)"

cd /opt/analise/SAMUEL
docker compose -f docker-compose.prod.yml up -d --wait

npm ci
npm run db:generate
npm run db:deploy

API_PROXY_TARGET=http://127.0.0.1:3469 NODE_OPTIONS=--max-old-space-size=1536 npm run web:build
npm run api:build

if pm2 restart analise-api analise-web --update-env; then
  true
else
  pm2 start ecosystem.config.cjs
  pm2 save
fi

curl -sS http://127.0.0.1:3468/api/v1/health
echo
pm2 status
