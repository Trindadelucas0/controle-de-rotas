generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

enum CompanyStatus {
  ACTIVE
  INACTIVE
}

enum UserRole {
  ADMIN
  MANAGER
  SUPERVISOR
  EMPLOYEE
}

enum UserStatus {
  ACTIVE
  INACTIVE
  SUSPENDED
}

enum EmployeeStatus {
  ACTIVE
  INACTIVE
  ON_LEAVE
  SUSPENDED
}

enum VehicleStatus {
  AVAILABLE
  IN_USE
  MAINTENANCE
  INACTIVE
}

enum CustomerStatus {
  ACTIVE
  INACTIVE
}

enum LocationStatus {
  OK
  PENDING
  FAILED
}

model Company {
  id             String         @id @default(uuid()) @db.Uuid
  name           String         @db.VarChar(200)
  tradeName      String?        @map("trade_name") @db.VarChar(200)
  document       String?        @db.VarChar(32)
  phone          String?        @db.VarChar(40)
  email          String?        @db.VarChar(255)
  address        String?        @db.VarChar(500)
  latitude       Float?
  longitude      Float?
  /// PostGIS Point SRID 4326 â€” mantido por trigger a partir de lat/lng
  location       Unsupported("geometry(Point,4326)")?
  locationStatus LocationStatus @default(PENDING) @map("location_status")
  status         CompanyStatus  @default(ACTIVE)
  createdAt      DateTime       @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt      DateTime       @updatedAt @map("updated_at") @db.Timestamptz(6)

  users      User[]
  employees  Employee[]
  customers  Customer[]
  vehicles   Vehicle[]
  auditLogs  AuditLog[]

  @@map("companies")
}

model User {
  id           String     @id @default(uuid()) @db.Uuid
  companyId    String     @map("company_id") @db.Uuid
  name         String     @db.VarChar(200)
  email        String     @unique @db.VarChar(255)
  passwordHash String     @map("password_hash")
  role         UserRole
  status       UserStatus @default(ACTIVE)
  lastLoginAt  DateTime?  @map("last_login_at") @db.Timestamptz(6)
  createdAt    DateTime   @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt    DateTime   @updatedAt @map("updated_at") @db.Timestamptz(6)

  company             Company              @relation(fields: [companyId], references: [id])
  employee            Employee?
  refreshTokens       RefreshToken[]
  passwordResetTokens PasswordResetToken[]
  auditLogs           AuditLog[]

  @@index([companyId])
  @@map("users")
}

model Employee {
  id           String         @id @default(uuid()) @db.Uuid
  companyId    String         @map("company_id") @db.Uuid
  userId       String?        @unique @map("user_id") @db.Uuid
  name         String         @db.VarChar(200)
  phone        String?        @db.VarChar(40)
  email        String?        @db.VarChar(255)
  jobTitle     String?        @map("job_title") @db.VarChar(120)
  registration String?        @db.VarChar(80)
  specialties  String[]       @default([])
  region       String?        @db.VarChar(120)
  status       EmployeeStatus @default(ACTIVE)
  createdAt    DateTime       @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt    DateTime       @updatedAt @map("updated_at") @db.Timestamptz(6)

  company Company @relation(fields: [companyId], references: [id])
  user    User?   @relation(fields: [userId], references: [id])

  @@index([companyId])
  @@index([status])
  @@map("employees")
}

model Customer {
  id             String         @id @default(uuid()) @db.Uuid
  companyId      String         @map("company_id") @db.Uuid
  name           String         @db.VarChar(200)
  tradeName      String?        @map("trade_name") @db.VarChar(200)
  document       String?        @db.VarChar(32)
  phone          String?        @db.VarChar(40)
  whatsapp       String?        @db.VarChar(40)
  email          String?        @db.VarChar(255)
  street         String?        @db.VarChar(200)
  number         String?        @db.VarChar(40)
  complement     String?        @db.VarChar(120)
  district       String?        @db.VarChar(120)
  city           String?        @db.VarChar(120)
  state          String?        @db.VarChar(2)
  zipCode        String?        @map("zip_code") @db.VarChar(16)
  latitude       Float?
  longitude      Float?
  /// PostGIS Point SRID 4326 â€” mantido por trigger a partir de lat/lng
  location       Unsupported("geometry(Point,4326)")?
  locationStatus LocationStatus @default(PENDING) @map("location_status")
  category       String?        @db.VarChar(80)
  priority       String?        @db.VarChar(40)
  notes          String?
  status         CustomerStatus @default(ACTIVE)
  createdAt      DateTime       @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt      DateTime       @updatedAt @map("updated_at") @db.Timestamptz(6)

  company Company @relation(fields: [companyId], references: [id])

  @@index([companyId])
  @@index([name])
  @@index([document])
  @@index([status])
  @@map("customers")
}

model Vehicle {
  id              String        @id @default(uuid()) @db.Uuid
  companyId       String        @map("company_id") @db.Uuid
  plate           String        @db.VarChar(16)
  brand           String?       @db.VarChar(80)
  model           String?       @db.VarChar(80)
  year            Int?
  fuelType        String?       @map("fuel_type") @db.VarChar(40)
  avgConsumption  Float?        @map("avg_consumption")
  capacity        Float?
  odometerKm      Float?        @map("odometer_km")
  status          VehicleStatus @default(AVAILABLE)
  createdAt       DateTime      @default(now()) @map("created_at") @db.Timestamptz(6)
  updatedAt       DateTime      @updatedAt @map("updated_at") @db.Timestamptz(6)

  company Company @relation(fields: [companyId], references: [id])

  @@unique([companyId, plate])
  @@index([companyId])
  @@index([status])
  @@map("vehicles")
}

model RefreshToken {
  id        String    @id @default(uuid()) @db.Uuid
  userId    String    @map("user_id") @db.Uuid
  tokenHash String    @map("token_hash")
  expiresAt DateTime  @map("expires_at") @db.Timestamptz(6)
  revokedAt DateTime? @map("revoked_at") @db.Timestamptz(6)
  userAgent String?   @map("user_agent")
  ip        String?   @db.VarChar(64)
  createdAt DateTime  @default(now()) @map("created_at") @db.Timestamptz(6)

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([tokenHash])
  @@map("refresh_tokens")
}

model PasswordResetToken {
  id        String    @id @default(uuid()) @db.Uuid
  userId    String    @map("user_id") @db.Uuid
  tokenHash String    @map("token_hash")
  expiresAt DateTime  @map("expires_at") @db.Timestamptz(6)
  usedAt    DateTime? @map("used_at") @db.Timestamptz(6)
  createdAt DateTime  @default(now()) @map("created_at") @db.Timestamptz(6)

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@index([userId])
  @@index([tokenHash])
  @@map("password_reset_tokens")
}

model AuditLog {
  id        String   @id @default(uuid()) @db.Uuid
  companyId String?  @map("company_id") @db.Uuid
  userId    String?  @map("user_id") @db.Uuid
  action    String   @db.VarChar(100)
  entity    String   @db.VarChar(100)
  entityId  String?  @map("entity_id") @db.Uuid
  metadata  Json?
  ip        String?  @db.VarChar(64)
  userAgent String?  @map("user_agent")
  createdAt DateTime @default(now()) @map("created_at") @db.Timestamptz(6)

  company Company? @relation(fields: [companyId], references: [id])
  user    User?    @relation(fields: [userId], references: [id])

  @@index([companyId])
  @@index([userId])
  @@index([action])
  @@map("audit_logs")
}
